## 关键代码示例

### 生成对战表算法（核心）

```javascript
// 单败淘汰赛生成算法
function generateBracket(students) {
  // 1. 随机打乱学员顺序
  const shuffled = students.sort(() => Math.random() - 0.5);
  
  // 2. 计算轮次（2的幂次）
  const rounds = Math.ceil(Math.log2(students.length));
  const totalSlots = Math.pow(2, rounds);
  
  // 3. 生成第一轮对战（包含轮空处理）
  const brackets = [];
  for (let i = 0; i < totalSlots; i += 2) {
    const s1 = shuffled[i] || null;
    const s2 = shuffled[i + 1] || null;
    
    brackets.push({
      round: 1,
      student1_id: s1?._id,
      student1_name: s1?.name,
      student2_id: s2?._id,
      student2_name: s2?.name,
      winner_id: s2 ? null : s1?._id // 轮空自动晋级
    });
  }
  
  // 4. 生成后续轮次空对战（待填充）
  for (let round = 2; round <= rounds; round++) {
    const matchCount = Math.pow(2, rounds - round);
    for (let i = 0; i < matchCount; i++) {
      brackets.push({
        round: round,
        student1_id: null,
        student1_name: '待定',
        student2_id: null,
        student2_name: '待定',
        winner_id: null
      });
    }
  }
  
  return brackets;
}
```

### 对战表晋级逻辑

```javascript
// 当录入比赛结果后，自动将获胜者填入下一轮
function updateMatchAndAdvance(matchId, winnerId) {
  // 1. 更新当前比赛结果
  const match = await db.collection('competition_brackets')
    .doc(matchId)
    .update({ winner_id: winnerId });
  
  // 2. 查找下一轮对应位置
  const currentMatch = await db.collection('competition_brackets')
    .doc(matchId)
    .get();
  
  const nextRound = currentMatch.round + 1;
  const nextPosition = Math.floor(currentMatch.match_no / 2);
  const isFirstStudent = currentMatch.match_no % 2 === 1;
  
  // 3. 更新下一轮对战表
  await db.collection('competition_brackets')
    .where({
      competition_id: currentMatch.competition_id,
      round: nextRound,
      match_no: nextPosition
    })
    .update({
      [isFirstStudent ? 'student1_id' : 'student2_id']: winnerId
    });
}
```

### 2. 比赛状态自动计算
```javascript
function getCompetitionStatus(startDate, endDate) {
  const now = new Date();
  const start = new Date(startDate);
  const end = new Date(endDate);
  
  if (now < start) return 'pending';
  if (now > end) return 'finished';
  return 'ongoing';
}
```
